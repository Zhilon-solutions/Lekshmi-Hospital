// GOOGLE APPS SCRIPT CODE
// Copy this code to your Google Apps Script editor

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName('Enquiries');
    
    // If sheet doesn't exist, create it with headers
    if (!sheet) {
      sheet = ss.insertSheet('Enquiries');
      sheet.appendRow(['Timestamp', 'Name', 'Phone', 'Subject', 'Message']);
      
      // Format header row
      var headerRange = sheet.getRange(1, 1, 1, 5);
      headerRange.setFontWeight('bold');
      headerRange.setBackground('#4285f4');
      headerRange.setFontColor('#ffffff');
      
      // Set column widths
      sheet.setColumnWidth(1, 150); // Timestamp
      sheet.setColumnWidth(2, 150); // Name
      sheet.setColumnWidth(3, 120); // Phone
      sheet.setColumnWidth(4, 200); // Subject
      sheet.setColumnWidth(5, 300); // Message
    }
    
    // Get form data
    var name = e.parameter.name || '';
    var phone = e.parameter.phone || '';
    var subject = e.parameter.subject || '';
    var message = e.parameter.message || '';
    var timestamp = new Date();
    
    // Append data to sheet
    sheet.appendRow([timestamp, name, phone, subject, message]);
    
    // Return success response
    return ContentService.createTextOutput(JSON.stringify({
      'result': 'success',
      'message': 'Enquiry submitted successfully'
    })).setMimeType(ContentService.MimeType.JSON);
    
  } catch(error) {
    return ContentService.createTextOutput(JSON.stringify({
      'result': 'error',
      'message': error.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  return ContentService.createTextOutput("Google Apps Script is running");
}
